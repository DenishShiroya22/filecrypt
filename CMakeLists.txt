cmake_minimum_required(VERSION 3.16)
project(filecrypt VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(FILECRYPT_ASAN "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(FILECRYPT_TSAN "Enable ThreadSanitizer" OFF)
option(FILECRYPT_BUILD_TESTS "Build dependency-free tests" ON)
if(FILECRYPT_ASAN AND FILECRYPT_TSAN)
    message(FATAL_ERROR "ASan and TSan require SEPARATE builds; they cannot be combined")
endif()
find_package(Threads REQUIRED)
add_library(filecrypt_options INTERFACE)
if(MSVC)
    target_compile_options(filecrypt_options INTERFACE /W4 /permissive- $<$<CONFIG:Release>:/O2>)
else()
    target_compile_options(filecrypt_options INTERFACE -Wall -Wextra -Wpedantic
        -Wconversion -Wshadow $<$<CONFIG:Release>:-O3>)
endif()
if(FILECRYPT_ASAN OR FILECRYPT_TSAN)
    if(MSVC OR WIN32 OR NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        message(FATAL_ERROR "These sanitizer configurations require GCC/Clang on Linux or a supported Unix host")
    endif()
    if(FILECRYPT_ASAN)
        set(FILECRYPT_SANITIZER address,undefined)
    else()
        set(FILECRYPT_SANITIZER thread)
    endif()
    # The last optimization flag wins over Release -O3 for useful diagnostics.
    target_compile_options(filecrypt_options INTERFACE
        -O1 -g -fno-omit-frame-pointer -fsanitize=${FILECRYPT_SANITIZER})
    target_link_options(filecrypt_options INTERFACE -fsanitize=${FILECRYPT_SANITIZER})
endif()
add_library(filecrypt_core src/crypto.cpp src/file_io.cpp src/pipeline.cpp src/thread_pool.cpp)
target_include_directories(filecrypt_core PUBLIC include)
target_link_libraries(filecrypt_core PUBLIC Threads::Threads filecrypt_options)
add_executable(filecrypt src/main.cpp)
target_link_libraries(filecrypt PRIVATE filecrypt_core)
if(FILECRYPT_BUILD_TESTS)
    enable_testing()
    add_executable(filecrypt_tests tests/tests.cpp)
    target_link_libraries(filecrypt_tests PRIVATE filecrypt_core)
    add_test(NAME filecrypt_tests COMMAND filecrypt_tests)
    set_tests_properties(filecrypt_tests PROPERTIES TIMEOUT 120)
endif()
install(TARGETS filecrypt RUNTIME DESTINATION bin)
